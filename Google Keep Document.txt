<script>
function calculate() {


  const n = id =>
    Math.max(
      0,
      Number(document.getElementById(id).value) || 0
    );


  const money = value =>
    value.toLocaleString("en-LK", {
      minimumFractionDigits: 2,
      maximumFractionDigits: 2
    });


  const A = n("A");
  const B = n("B");
  const C = n("C");
  const D = n("D");


  const covers = [
    ["FIRE", ((A + B) / 100) * 0.3, "fire"],
    ["RCC", ((A + B) / 100) * 0.05, "rcc"],
    ["BURGLARY", ((A + B) / 100) * 0.175, "burglary"],
    ["ELECTRIC ITEMS", (C / 100) * 0.5, "electric"],
    ["GLASSES", (D / 100) * 3, "glasses"],
    ["TC", ((A + B) / 100) * 0.0375, "tc"]
  ];


  let sum = 0;
  let html = "";


  covers.forEach(item => {
    const [name, amount, id] = item;


    if (document.getElementById(id).checked) {
      sum += amount;


      html += `
        <tr>
          <td>${name}</td>
          <td class="money">${money(amount)}</td>
        </tr>
      `;
    }
  });


  const K = (sum / 100) * 3;
  const L = 300;
  const M = (sum / 100) * 18;
  const total = sum + K + L + M;


  html += `
    <tr>
      <td>Cover Premium Subtotal</td>
      <td class="money">${money(sum)}</td>
    </tr>


    <tr>
      <td>ADMIN FEE</td>
      <td class="money">${money(K)}</td>
    </tr>


    <tr>
      <td>POLICY FEE</td>
      <td class="money">${money(L)}</td>
    </tr>


    <tr>
      <td>VAT</td>
      <td class="money">${money(M)}</td>
    </tr>


    <tr class="total">
      <td>TOTAL</td>
      <td class="money">${money(total)}</td>
    </tr>
  `;


  document.getElementById("rows").innerHTML = html;


  document.getElementById("businessOut").textContent =
    document.getElementById("business").value.trim() || "—";


  document.getElementById("dateOut").textContent =
    new Date().toLocaleDateString("en-GB");
}


document.getElementById("dateOut").textContent =
  new Date().toLocaleDateString("en-GB");


calculate();
</script>
